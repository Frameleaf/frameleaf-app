cmake_minimum_required(VERSION 3.16)
project(frameleaf_image_hdr LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HEIF REQUIRED IMPORTED_TARGET libheif)
find_package(EXPAT REQUIRED)
find_path(UHDR_INCLUDE_DIR ultrahdr_api.h REQUIRED)
find_library(UHDR_LIBRARY uhdr REQUIRED)
find_path(NODE_API_INCLUDE_DIR node_api.h PATHS /usr/include/node /usr/local/include/node REQUIRED)
add_library(image-hdr MODULE image-hdr.cc)
target_compile_features(image-hdr PRIVATE cxx_std_17)
target_compile_options(image-hdr PRIVATE -Wall -Wextra -Werror)
target_include_directories(image-hdr PRIVATE ${UHDR_INCLUDE_DIR} ${NODE_API_INCLUDE_DIR})
target_link_libraries(image-hdr PRIVATE PkgConfig::HEIF EXPAT::EXPAT ${UHDR_LIBRARY})
set_target_properties(image-hdr PROPERTIES PREFIX "" SUFFIX ".node")
if(APPLE)
  target_link_options(image-hdr PRIVATE -undefined dynamic_lookup)
endif()
install(TARGETS image-hdr LIBRARY DESTINATION lib/frameleaf)
